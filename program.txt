function bubbleSort(arr : num[]) : num[] {
    let n : num = 1000;
    let i : num = 0;
    while (i < n) {
        let j : num = 0;
        while (j < n - 1) {
            if (arr[j] > arr[j + 1]) {
                let temp : num = arr[j];
                arr[j] = arr[j + 1];
                arr[j + 1] = temp;
            }
            j = j + 1;
        }
        i = i + 1;
    }
    return arr;
}

let data: num[] = [-10, -20, 10];
let result: num[] = bubbleSort(data);
