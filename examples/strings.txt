let greeting: string = "Hello";
const target: string = "World";
const punctuation: string = "!";

function greet(person: string): string {
    let full_greeting: string = greeting + ", " + person + punctuation;
    print(full_greeting);
    return full_greeting;
}

let combined: string = greeting + " " + target;
print(combined);

let greeted_person: string = greet(person_name);
