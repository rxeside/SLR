function bubbleSort(arr: num[]): num[]
{
    let size: num =  arrayLength(arr);
    let i: num = 0;
    let j: num = 0;
    let temp: num = 0;

    while (i < size) {
        j = 0;
        while (j < (size - i) - 1) {
            if (arr[j] > arr[j + 1])
            {
                temp = arr[j];
                arr[j] = arr[j + 1];
                arr[j + 1] = temp;
            }
            j = j + 1;
            print(j);
        }
        i = i + 1;
    }

    return arr;
}
