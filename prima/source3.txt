function isPrime(num: number): bool
{
    if (num < 2)
    {
        return false;
    }

    if (num == 2)
    {
        return true;
    }

    if (num % 2 == 0)
    {
        return false;
    }

    for (var i: number = 3; i < sqrt(num) + 1; i = i + 2)
    {
        if (num % i == 0)
        {
            return false;
        }
    }

    return true;
}

const start: number = 10;
const end: number = 100;

for (var num: number = start; num < end + 1; num = num + 1)
{
    if (isPrime(num))
    {
        println(num);
    }
}
