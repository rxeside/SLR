function isPrime(num)
{
    if (num < 2)
    {
        return false;
    }

    if (num == 2)
    {
        return true;
    }

    if (num % 2 == 0)
    {
        return false;
    }

    for (var i = 3; i < sqrt(num) + 1; i = i + 2)
    {
        if (num % i == 0)
        {
            return false;
        }
    }

    return true;
}

const start = 10;
const end = 100;

for (var num = start; num < end + 1; num = num + 1)
{
    if (isPrime(num))
    {
        println(num);
    }
}
